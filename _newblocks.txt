    /**
     * 秒杀库存预扣结果 -> 业务错误码
     *
     * @param deductResult
     * @return
     */
    private ResultCodeEnum convertDeductResult(SeckillStockDeductResultEnum deductResult) {
        switch (deductResult) {
            case ACTIVITY_NOT_STARTED:
                return ResultCodeEnum.SECKILL_ACTIVITY_NOT_STARTED;
            case ACTIVITY_ENDED:
                return ResultCodeEnum.SECKILL_ACTIVITY_ENDED;
            case REPEATED_ORDER:
                return ResultCodeEnum.SECKILL_ORDER_DUPLICATE;
            case SOLD_OUT:
                return ResultCodeEnum.SECKILL_GOODS_SOLD_OUT;
            case STOCK_NOT_PREHEATED:
                return ResultCodeEnum.SECKILL_STOCK_NOT_PREHEATED;
            default:
                return ResultCodeEnum.SYSTEM_ERROR;
        }
